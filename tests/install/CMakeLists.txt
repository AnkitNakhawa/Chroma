cmake_minimum_required(VERSION 3.25)
project(chroma_c_only_consumer LANGUAGES C)
find_package(Chroma CONFIG REQUIRED)
add_executable(consumer main.c)
target_compile_features(consumer PRIVATE c_std_99)
target_link_libraries(consumer PRIVATE chroma::c)
if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(consumer PRIVATE -Wall -Wextra -Werror -pedantic -fshort-enums)
endif()
